    overtimePay = overtimeHours * 50
    bonus = 0

    IF overtimeHours > 20 THEN
        bonus = 500
    END IF

END

import java.util.Scanner;
public class MiniPosSystem {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int stockQty = 50;

        double itemPrice = input.nextDouble();
        int reqQty = input.nextInt();

        if (reqQty <= 0) {
            System.out.println("Error: Quantity must be > 0");
        } else if (reqQty <= stockQty) {
            double total = itemPrice * reqQty;
            if (total >= 1000) {
                total = total - (total * 0.10); // 10% Discount
                System.out.println("10% Discount Applied!");
            }
            stockQty = stockQty - reqQty;
            System.out.println("Final Total: " + total + " EGP");
            System.out.println("Remaining Stock: " + stockQty);
        } else {
            System.out.println("Sorry, Out of Stock!");
        }
    }
}

System.out.println("Select Payment: [1] Cash  [2] Card  [3] Wallet");

switch (paymentMethod) {

متى نفضل switch على if-else؟

  • عندما نختبر متغيراً واحداً فقط مقابل قيم محددة وثابتة Discrete Values (مثل أرقام القوائم Menus، طرق الدفع، أو نوع العضوية).
  • أهمية break;: أخذ استراحة والخروج من البلوك فور تطابق الحالة.
  • أهمية default:: تعمل مثل else في حال إدخال رقم غير موجود في الاختيارات.

import java.util.Scanner;

التمرين العملي الثاني: نظام خصومات العملاء (CRM & Sales Discount Lab)

Hands-on Lab (20 دقيقة)

🛠️ المطلوب تنفيذه باستخدام switch & if-else:

  • المدخلات: إجمالي المشتريات double amount، ونوع العضوية int type.
  • قواعد الخصم:
    • [1] Regular: خصم 5% إذا كانت المشتريات >= 500، وإلا 0%.
    • [2] Silver: خصم 10%، ويصبح 15% إذا كانت المشتريات >= 1000.
    • [3] Gold: خصم ثابت 20% على أي فاتورة.
    • غير ذلك: يطبع "Invalid Membership Type!". 
import java.util.Scanner;
public class CrmDiscountLab {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        System.out.print("Enter Order Amount: ");
        double amount = in.nextDouble();
        System.out.print("Membership [1] Regular [2] Silver [3] Gold: ");
        int type = in.nextInt();
        double discRate = 0.0;

        switch (type) {
            case 1:
                if (amount >= 500) discRate = 0.05;
                break;
            case 2:
                discRate = (amount >= 1000) ? 0.15 : 0.10;
                break;
            case 3:
                discRate = 0.20;
                break;
            default:
                System.out.println("Invalid Membership Type!");
                return;
        }
        double finalPay = amount - (amount * discRate);
        System.out.println("Final Amount to Pay: " + finalPay + " EGP");
    }
}

🛠️ المطلوب تنفيذه في التحدي:

  • المدخلات: اسم الصنف، الرصيد الحالي currentStock، حد الأمان minThreshold، نوع الحركة ([1] Sell/Issue أو [2] Restock/Add)، والكمية qty.
  • القواعد:
    1. التأكد أن qty > 0.
    2. في حالة البيع [1]: إذا كانت الكمية متوفرة نخصمها، ثم نفحص: إذا أصبح الرصيد المتبقي <= minThreshold نطبع تحذيراً: "WARNING: Low Stock! Reorder soon.".
    3. في حالة التوريد [2]: نجمع الكمية على الرصيد ونطبع الرصيد المحدث. 
import java.util.Scanner;
public class SmartInventoryLab {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        int currentStock = 25, minThreshold = 10;
        System.out.print("Operation [1] Sell [2] Restock: ");
        int op = in.nextInt();
        System.out.print("Enter Quantity: ");
        int qty = in.nextInt();

        if (qty <= 0) {
            System.out.println("Error: Quantity must be positive!");
        } else if (op == 1) {
            if (qty <= currentStock) {
                currentStock -= qty;
                System.out.println("Sold! Remaining Stock: " + currentStock);
                if (currentStock <= minThreshold) {
                    System.out.println("⚠️ WARNING: Low Stock! Please reorder.");
                }
            } else {
                System.out.println("❌ Insufficient Stock!");
            }
        } else if (op == 2) {
            currentStock += qty;
            System.out.println("Restocked! New Stock: " + currentStock);
        }
    }
}
